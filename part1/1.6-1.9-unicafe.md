import { useState } from "react";

const Button = ({ onClick, text }) => {
  return <button onClick={onClick}>{text}</button>;
};

const StatisticLine = ({ text, value }) => {
  return (
    <tr>
      <th>{text}</th>
      <td>{value}</td>
    </tr>
  );
};

const Statistics = ({ average, percentage, feedback, total }) => {
  return (
    <>
      <h2>Statistics:</h2>

      {total ? (
        <table>
          <tbody>
            <StatisticLine text={"Good"} value={feedback.good} />
            <StatisticLine text={"Bad"} value={feedback.bad} />
            <StatisticLine text={"Neutral"} value={feedback.neutral} />
            <StatisticLine text={"Total"} value={total} />
            <StatisticLine text={"Average"} value={average} />
            <StatisticLine text={"Percentage"} value={`${percentage} %`} />
          </tbody>
        </table>
      ) : (
        <p>No feedback given yet</p>
      )}
    </>
  );
};

const App = () => {
  const [feedback, setFeedback] = useState({
    bad: 0,
    good: 0,
    neutral: 0,
  });

  const handleGood = () => {
    setFeedback((prev) => ({ ...prev, good: prev.good + 1 }));
  };

  const handleBad = () => {
    setFeedback((prev) => ({ ...prev, bad: prev.bad + 1 }));
  };

  const handleNeutral = () => {
    setFeedback((prev) => ({ ...prev, neutral: prev.neutral + 1 }));
  };

  const total = feedback.good + feedback.bad + feedback.neutral;

  const feedbackScores = {
    good: 1,
    bad: -1,
    neutral: 0,
  };

  const goodScore = feedbackScores.good * feedback.good;
  const badScore = feedbackScores.bad * feedback.bad;
  const neutralScore = feedbackScores.neutral * feedback.neutral;
  const average = (goodScore + badScore + neutralScore) / total;
  const goodPercentage = (feedback.good / total) * 100;

  return (
    <div>
      <h2>Give Feedback</h2>
      <Button onClick={handleGood} text={"good"}></Button>
      <Button onClick={handleNeutral} text={"neutral"}></Button>
      <Button onClick={handleBad} text={"bad"}></Button>
      <Statistics
        average={average}
        percentage={goodPercentage}
        feedback={feedback}
        total={total}
      />
    </div>
  );
};

export default App;
